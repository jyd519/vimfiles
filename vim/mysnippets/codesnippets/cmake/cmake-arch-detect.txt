# Use the target architecture, including cross-compilation and Win32 builds.
if(NODE_ARCH)
  set(LOADER_ARCH "${NODE_ARCH}")
elseif(CMAKE_OSX_ARCHITECTURES)
  set(LOADER_ARCH "${CMAKE_OSX_ARCHITECTURES}")
elseif(CMAKE_CXX_COMPILER_ARCHITECTURE_ID)
  set(LOADER_ARCH "${CMAKE_CXX_COMPILER_ARCHITECTURE_ID}")
elseif(CMAKE_GENERATOR_PLATFORM)
  set(LOADER_ARCH "${CMAKE_GENERATOR_PLATFORM}")
else()
  set(LOADER_ARCH "${CMAKE_SYSTEM_PROCESSOR}")
endif()
string(TOLOWER "${LOADER_ARCH}" LOADER_ARCH)
if(LOADER_ARCH MATCHES "^(amd64|x86_64|x64)$")
  set(LOADER_ARCH "x64")
elseif(LOADER_ARCH MATCHES "^(win32|x86|i[3-6]86|ia32)$")
  set(LOADER_ARCH "ia32")
elseif(LOADER_ARCH MATCHES "^(aarch64|arm64)$")
  set(LOADER_ARCH "arm64")
elseif(LOADER_ARCH MATCHES "^(loongarch64|loong64)$")
  set(LOADER_ARCH "loong64")
endif()
# Keep macOS universal builds in a single directory.
string(REPLACE ";" "-" LOADER_ARCH "${LOADER_ARCH}")
set(LOADER_BIN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/bin/${LOADER_ARCH}")
