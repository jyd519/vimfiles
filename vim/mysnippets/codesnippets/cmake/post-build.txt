  set_target_properties(${NAME} PROPERTIES PREFIX "" SUFFIX ".node")
  add_custom_command(TARGET ${NAME} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${LOADER_BIN_DIR}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "$<TARGET_FILE:${NAME}>"
      "${LOADER_BIN_DIR}/${NAME}$<$<CONFIG:Debug>:d>.node"
    COMMENT "Copying ${NAME} to bin/${LOADER_ARCH}"
    VERBATIM
  )
